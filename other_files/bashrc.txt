# Définition des variables pour les tailles de colonnes
export PID_COL_SIZE=8
export NAME_COL_SIZE=32
export ST_COL_SIZE=2
export TIME_COL_SIZE=8
export NODES_COL_SIZE=4
export NODELIST_COL_SIZE=40

# Alias pour squeue -u $USER
alias sq='squeue -u $USER'

# Alias pour scancel
alias sc='scancel -u $USER'

# Alias pour squeue avec nom complet et PID
alias sqn='squeue -u $USER -h -o "%.'$NAME_COL_SIZE'j %.'$PID_COL_SIZE'i %.'$ST_COL_SIZE't %.'$TIME_COL_SIZE'M %.'$NODES_COL_SIZE'D %.'$NODELIST_COL_SIZE'R" | awk '\''BEGIN { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", "NAME", "PID", "ST", "TIME", "NODES", "NODELIST" } { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", $1, $2, $3, $4, $5, $6 }'\'''

# Alias pour squeue avec nom complet et PID des jobs running
alias sqnr='squeue -u $USER -h -o "%.'$NAME_COL_SIZE'j %.'$PID_COL_SIZE'i %.'$ST_COL_SIZE't %.'$TIME_COL_SIZE'M %.'$NODES_COL_SIZE'D %.'$NODELIST_COL_SIZE'R" | awk '\''BEGIN { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", "NAME", "PID", "ST", "TIME", "NODES", "NODELIST" } $3 == "R" { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", $1, $2, $3, $4, $5, $6 }'\'''

# Alias pour squeue avec nom complet et PID des jobs running ou pending pour raison autre que Dependency
alias sqna='squeue -u $USER -h -o "%.'$NAME_COL_SIZE'j %.'$PID_COL_SIZE'i %.'$ST_COL_SIZE't %.'$TIME_COL_SIZE'M %.'$NODES_COL_SIZE'D %.'$NODELIST_COL_SIZE'R" | awk '\''BEGIN { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", "NAME", "PID", "ST", "TIME", "NODES", "NODELIST" } $3 == "R" || ($3 == "PD" && $6 != "Dependency") { printf "%-'$NAME_COL_SIZE's %-'$PID_COL_SIZE's %-'$ST_COL_SIZE's %-'$TIME_COL_SIZE's %-'$NODES_COL_SIZE's %-'$NODELIST_COL_SIZE's\n", $1, $2, $3, $4, $5, $6 }'\'''

# Alias pour rm generation & slurm
alias rmres='rm -rf $HOME/Codethesis/generation/* && rm -rf $HOME/Codethesis/slurm_code/slurm_scripts/* && rm -rf $HOME/Codethesis/slurm_code/output_launch_slurm_jobs/*'
