GENERATION_DIR="$HOME/Codethesis/generation"

# Fonction pour gérer la suppression avec confirmation
rm_res() {
    # Rechercher les fichiers à supprimer dans le dossier GENERATION
    files_to_delete_gen=$(find "$GENERATION_DIR" -mindepth 1 \
        ! -name "results" ! -path "$GENERATION_DIR/results/*" \
        ! -name "plots" ! -path "$GENERATION_DIR/plots/*" \
        ! -name "experience.db" -print)
    
    if [ -z "$files_to_delete_gen" ]; then
        echo "Nothing to delete."
        return
    fi
    
    # Afficher la liste des fichiers à supprimer et demander confirmation
    echo "The following files and directories will be deleted:"
    echo "$files_to_delete_gen"
    
    echo "Proceed with deletion? (y/n)"
    read -r answer

    # Si l'utilisateur confirme, procéder à la suppression
    if [ "$answer" = "y" ]; then
        if [ ! -z "$files_to_delete_gen" ]; then
            find "$GENERATION_DIR" -mindepth 1 \
                ! -name "results" ! -path "$GENERATION_DIR/results/*" \
                ! -name "plots" ! -path "$GENERATION_DIR/plots/*" \
                ! -name "experience.db" -print | xargs rm -rf
        fi
        echo "Deletion completed."
    else
        echo "Deletion canceled."
    fi
}

alias rmres=rm_res

# Fonction pour supprimer le dossier plots
rm_plots() {
    PLOTS_DIR="$GENERATION_DIR/plots"

    if [ -d "$PLOTS_DIR" ]; then
        echo "Le dossier 'plots' va être supprimé. Confirmer? (y/n)"
        read -r answer
        if [ "$answer" = "y" ]; then
            rm -rf "$PLOTS_DIR"
            echo "Le dossier 'plots' a été supprimé."
        else
            echo "Suppression annulée."
        fi
    else
        echo "Le dossier 'plots' n'existe pas."
    fi
}

alias rmplots=rm_plots
alias rmplot=rm_plots

scancel_pending() {
    echo "Confirmez-vous l'annulation de tous les jobs en attente? (y/n)"
    read -r response
    if [[ "$response" == "y" ]]; then
        squeue -u $USER -h -t PENDING -o %i | xargs scancel
    else
        echo "Annulation annulée."
    fi
}

scancel_all() {
    echo "Confirmez-vous l'annulation de tous vos jobs? (y/n)"
    read -r response
    if [[ "$response" == "y" ]];then
        scancel -u $USER
    else
        echo "Annulation annulée."
    fi
}

alias scp=scancel_pending
alias sc=scancel_all

tail_print() {
    # Assurez-vous qu'il y a au moins trois arguments
    if [ "$#" -lt 3 ]; then
        echo "Usage: tail_print <line_count> <experience> <file_type>"
        return 1
    fi

    local line_count="$1"
    local experience="$2"
    local file_type="$3"
    local file_path="$HOME/Codethesis/generation/${experience}/slurm/output"

    # Déterminez le chemin en fonction du troisième argument
    case "$file_type" in
        taskset_* | assignment_* | scheduling_* | batch_taskset_* | batch_assignment_* | batch_scheduling_*)
            local category="${file_type%%_*}"
            file_path+="/${category}/${file_type}.txt"
            ;;
        full_pipeline)
            file_path+="/full_pipeline_${experience}.txt"
            ;;
        analyze_results)
            file_path+="/analyze_results.txt"
            ;;
        all_tasksets_master | all_assignments_master | all_schedulings_master)
            local category="${file_type%_*}"
            category="${category#all_}"
            # Adjusting the directory name to be singular
            if [[ $category == "tasksets" ]]; then
                category_dir="taskset"  
            fi
            
            if [[ $category == "assignments" ]]; then
                category_dir="assignment"  
            fi
            
            if [[ $category == "schedulings" ]]; then
                category_dir="scheduling"  
            fi
            file_path+="/${category_dir}/${category}_master.txt"
            ;;
        submit)
            file_path+="/submit_full_pipeline.txt"
            ;;
        *)
            echo "Invalid file type"
            return 1
            ;;
    esac

    # Exécuter tail sur le fichier
    tail -n "$line_count" "$file_path"
}


alias t20='tail_print 20'
