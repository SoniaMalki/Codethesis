generate_and_submit() {
    if [ -z "$1" ] || ! [[ "$1" =~ ^[0-9]+$ ]]; then
        echo "Erreur: L'argument 'index' est obligatoire et doit être un chiffre."
        echo "Utilisation: generate_and_submit <index>"
        return 1
    fi

    index="$1"
    exp_name="$2"
    output_file="generated_script.slurm"

    # Enregistrer le nom de l'expérience dans un fichier
    echo "$exp_name" > $HOME/.current_experience

    sbatch <<EOF
#!/bin/bash
#SBATCH --time=2-00:00:00
#SBATCH --output=./create_full_pipeline_output.txt
#SBATCH --job-name=create_full_pipeline

if [[ \$(hostname) == *"her" ]]; then
    module load Python/3.9.6-GCCcore-11.2.0
    module load GLPK/5.0-GCCcore-11.2.0
    module load Gurobi-Optimizer/9.5.1

elif [[ \$(hostname) == *"lm4" ]]; then
    module load releases/2023a
    module load Python/3.11.3-GCCcore-12.3.0
    module load GLPK/5.0-GCCcore-12.3.0
    module load tis/2018.01
    module load gurobi/gurobi1102

elif [[ \$(hostname) == *"nic" ]]; then
    module load releases/2022b
    module load Python/3.10.8-GCCcore-12.2.0
    module load GLPK/5.0-GCCcore-12.2.0
    module load Gurobi/10.0.3-GCCcore-12.2.0

elif [[ \$(hostname) == *"drg2" ]]; then
    module load Python/3.11.3-GCCcore-12.3.0
    export GLOBALSCRATCH=$HOME
fi

source $GLOBALSCRATCH/myenv/bin/activate

# Exécute le script Python pour générer les scripts SLURM
echo "Launching python generate_slurm_scripts"
echo "--------PYTHON-OUTPUT-BEGIN------------------"
python3 -u \$HOME/Codethesis/main.py generate_slurm_scripts $index $exp_name
echo "--------PYTHON-OUTPUT-END--------------------"
echo "Python generate slurm script finish."

# Soumet le script généré en tant que job
echo "Launching fullpipeline"
echo "--------SBATCH-OUTPUT-BEGIN------------------"

# Utiliser un fichier temporaire pour capturer stdout et stderr
tempfile=\$(mktemp)

# Soumission du job et capture de la sortie et des erreurs
sbatch "\$GENERATION_DIR/${exp_name}_${index}/slurm/master/full_pipeline_${exp_name}.slurm" >"\$tempfile" 2>&1
output=\$(cat "\$tempfile")

# Vérifier s'il y a une erreur dans la sortie
if echo "\$output" | grep -q "error"; then
    # Afficher l'erreur et quitter
    echo "Job not launched. Output:"
    echo "\$output"
else
    # Extraire le Job ID de la sortie de sbatch
    job_id=\$(echo "\$output" | awk '{print \$4}')

    # Vérifier si le Job ID est valide
    if [[ -n "\$job_id" ]]; then
        # Extraire le nom du job en utilisant scontrol
        job_name=\$(scontrol show job "\$job_id" | grep "JobName=" | awk -F= '{print \$3}')

        # Afficher le message original et le nom du job
        echo "\$output | Job Name: \$job_name"
    else
        # Si le job n'a pas été lancé, afficher un message d'erreur
        echo "Job not launched. Output:"
        echo "\$output"
    fi
fi

# Nettoyer le fichier temporaire
rm -f "\$tempfile"
echo "--------SBATCH-OUTPUT-END------------------"

EOF

}

tail_print() {
    # Assurez-vous qu'il y a au moins trois arguments
    if [ "$#" -lt 3 ]; then
        echo "Usage: tail_print <line_count> <experience> <file_type>"
        return 1
    fi

    local line_count="$1"
    local experience="$2"
    local file_type="$3"

    # Si l'argument 'experience' est 'expn', remplacer par le contenu de .current_experience
    if [ "$experience" == "expn" ]; then
        experience=$(cat $HOME/.current_experience)
    fi

    local file_path="$GENERATION_DIR/${experience}/slurm/output"

    # Déterminez le chemin en fonction du troisième argument
    case "$file_type" in
        batch_taskset_* | batch_assignment_* | batch_scheduling_*)
            local category="${file_type#batch_}"
            category="${category%%_*}"
            file_path+="/${category}/${file_type}.txt"
            ;;
        taskset_* | assignment_* | scheduling_*)
            local category="${file_type%%_*}"
            file_path+="/${category}/${file_type}.txt"
            ;;
        full_pipeline)
            file_path+="/full_pipeline_${experience}.txt"
            ;;
        analyze_results)
            file_path+="/analyze_results.txt"
            ;;
        all_tasksets_master | all_assignments_master | all_schedulings_master)
            local category="${file_type%_*}"
            category="${category#all_}"
            # Adjusting the directory name to be singular
            if [[ $category == "tasksets" ]]; then
                category_dir="taskset"  
            fi
            
            if [[ $category == "assignments" ]]; then
                category_dir="assignment"  
            fi
            
            if [[ $category == "schedulings" ]]; then
                category_dir="scheduling"  
            fi
            file_path+="/${category_dir}/${category}_master.txt"
            ;;
        submit)
            file_path+="/submit_full_pipeline.txt"
            ;;
        *)
            echo "Invalid file type"
            return 1
            ;;
    esac

    # Exécuter tail sur le fichier
    tail -n "$line_count" "$file_path"
}



# Fonction pour gérer la suppression avec confirmation
rm_res() {
    # Create the empty directory if it doesn't exist
    if [ ! -d "$EMPTY_DIR" ]; then
        mkdir -p "$EMPTY_DIR"
    fi

    # Rechercher les fichiers à supprimer dans le dossier GENERATION
    files_to_delete_gen=$(find "$GENERATION_DIR" -mindepth 1 \
        ! -name "results*" ! -path "$GENERATION_DIR/results_*/*" \
        ! -name "experience*.db" \
        ! -name "plots*" ! -path "$GENERATION_DIR/plots_*/*" \
        ! -name "experience.db" -print)
    
    if [ -z "$files_to_delete_gen" ]; then
        echo "Nothing to delete."
        return
    fi
    
    # Afficher la liste des fichiers à supprimer et demander confirmation
    echo "The following files and directories will be deleted:"
    echo "$files_to_delete_gen"
    
    echo "Proceed with deletion? (y/n)"
    read -r answer

    # Si l'utilisateur confirme, procéder à la suppression
    if [ "$answer" = "y" ]; then
        rsync -a --delete-before \
            --exclude="results*" \
            --exclude="results*/*" \
            --exclude="plots*" \
            --exclude="plots*/*" \
            --exclude="experience*.db" \
            "$EMPTY_DIR/" "$GENERATION_DIR/"
        echo "Deletion completed."
    else
        echo "Deletion canceled."
    fi
}

# Fonction pour supprimer le dossier plots
rm_plots() {
    if [ -z "$1" ] || ! [[ "$1" =~ ^[0-9]+$ ]]; then
        echo "Erreur: L'argument 'index' est obligatoire et doit être un chiffre."
        echo "Utilisation: rm_plots <index>"
        return 1
    fi

    index="$1"
    PLOTS_DIR="$GENERATION_DIR/plots_$index"

    # Create the empty directory if it doesn't exist
    if [ ! -d "$EMPTY_DIR" ]; then
        mkdir -p "$EMPTY_DIR"
    fi

    if [ -d "$PLOTS_DIR" ]; then
        echo "Le dossier '$PLOTS_DIR' va être supprimé. Confirmer? (y/n)"
        read -r answer
        if [ "$answer" = "y" ]; then
            rsync -a --delete-before "$EMPTY_DIR/" "$PLOTS_DIR/"
            rmdir "$PLOTS_DIR" 2>/dev/null || true
            echo "Le dossier 'plots' a été supprimé."
        else
            echo "Suppression annulée."
        fi
    else
        echo "Le dossier '$PLOTS_DIR' n'existe pas."
    fi
}


# Fonction pour supprimer le fichier experience.db
rm_db() {
    if [ -z "$1" ] || ! [[ "$1" =~ ^[0-9]+$ ]]; then
        echo "Erreur: L'argument 'index' est obligatoire et doit être un chiffre."
        echo "Utilisation: rm_db <index>"
        return 1
    fi

    index="$1"
    DB_FILE="$GENERATION_DIR/experience_$index.db"

    if [ -f "$DB_FILE" ];then
        echo "Le fichier '$DB_FILE' va être supprimé. Confirmer? (y/n)"
        read -r answer
        if [ "$answer" = "y" ];then
            rm -f "$DB_FILE"
            echo "Le fichier '$DB_FILE' a été supprimé."
        else
            echo "Suppression annulée."
        fi
    else
        echo "Le fichier '$DB_FILE' n'existe pas."
    fi
}

# Fonction pour supprimer le dossier generation
rm_gen() {
    # Créer le répertoire vide s'il n'existe pas
    if [ ! -d "$EMPTY_DIR" ]; then
        mkdir -p "$EMPTY_DIR"
    fi

    if [ -d "$GENERATION_DIR" ]; then
        echo "Le dossier 'generation' va être supprimé avec tout son contenu. Confirmer? (y/n)"
        read -r answer
        if [ "$answer" = "y" ]; then
            # Utilisation de rsync pour vider rapidement le dossier
            rsync -a --delete-before --ignore-existing "$EMPTY_DIR/" "$GENERATION_DIR/"
            rm -rf "$GENERATION_DIR"
            echo "Le dossier 'generation' a été supprimé."
        else
            echo "Suppression annulée."
        fi
    else
        echo "Le dossier 'generation' n'existe pas."
    fi
}




scancel_pending() {
    echo "Confirmez-vous l'annulation de tous les jobs en attente? (y/n)"
    read -r response
    if [[ "$response" == "y" ]]; then
        squeue -u $USER -h -t PENDING -o %i | xargs scancel
    else
        echo "Annulation annulée."
    fi
}

scancel_all() {
    echo "Confirmez-vous l'annulation de tous vos jobs? (y/n)"
    read -r response
    if [[ "$response" == "y" ]];then
        scancel -u $USER
    else
        echo "Annulation annulée."
    fi
}
