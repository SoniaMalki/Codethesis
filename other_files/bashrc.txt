# Alias pour rm generation & slurm
# Variables pour les chemins et exclusions
GENERATION_DIR="$HOME/Codethesis/generation"
SLURM_SCRIPTS_DIR="$HOME/Codethesis/slurm_code/slurm_scripts"
OUTPUT_LAUNCH_DIR="$HOME/Codethesis/slurm_code/output_launch_slurm_jobs"

# Fonction pour gérer la suppression avec confirmation
rm_res() {
    # Rechercher les fichiers à supprimer, en excluant certains dossiers et fichiers
    files_to_delete=$(find "$GENERATION_DIR" -mindepth 1 \
        ! -name "results" ! -path "$GENERATION_DIR/results/*" \
        ! -name "plots" ! -path "$GENERATION_DIR/plots/*" \
        ! -name "experience.db" -print)
    
    # Vérifier si la liste des fichiers à supprimer est vide
    if [ -z "$files_to_delete" ]; then
        echo "No files or directories to delete."
        return
    fi

    # Afficher la liste des fichiers à supprimer et demander confirmation
    echo "The following files and directories will be deleted:"
    echo "$files_to_delete"
    echo "Proceed with deletion? (y/n)"
    read -r answer

    # Si l'utilisateur confirme, procéder à la suppression
    if [ "$answer" = "y" ]; then
        echo "$files_to_delete" | xargs rm -rf
        rm -rf "$SLURM_SCRIPTS_DIR"/*
        rm -rf "$OUTPUT_LAUNCH_DIR"/*
        echo "Deletion completed."
    else
        echo "Deletion canceled."
    fi
}

alias rmres=rm_res
